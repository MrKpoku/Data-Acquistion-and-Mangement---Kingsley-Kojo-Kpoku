RUNNING MY SQL QUERY FROM R

movie_data <- dbGetQuery(con, "
     SELECT
        us.person_id,
        us.person_name,
        rat.movie_id,
        rat.rating,
        mov.title,
        mov.genre
    FROM users AS us
    INNER JOIN ratings AS rat
        ON us.person_id = rat.person_id
    INNER JOIN movies AS mov
        ON rat.movie_id = mov.movie_id
")

1A.HIGHEST AVERAGE RATINGS

average_rating <- movie_data %>%
  group_by(title) %>%
  summarise(
    average_rating = mean(rating, na.rm = TRUE)
  ) %>%
  arrange(desc(average_rating))

movie_analysis <- movie_data %>%
  group_by(title) %>%
  summarise(
    number_of_ratings = sum(!is.na(rating)),
